usage: simplicio-py scratch [-h] [--stack STACK] [--name NAME] [--dest DEST]
                            [--root ROOT] [--planner PLANNER] [--plan-only]
                            [--skip-install] [--list-stacks]
                            [--show-stack SLUG] [--list-recipes]
                            [--slot KEY=VALUE] [--json]
                            [goal]

positional arguments:
  goal               one-line description of the project to create

options:
  -h, --help         show this help message and exit
  --stack STACK      explicit stack slug (e.g. py-fastapi); inferred if
                     omitted
  --name NAME        project directory name (derived from goal if omitted)
  --dest DEST        parent directory where the project dir is created
                     (default: cwd)
  --root ROOT        existing project root used for stack detection (default:
                     cwd)
  --planner PLANNER  override SIMPLICIO_PLANNER for this run only
  --plan-only        run planner and print the validated plan; skip execution
  --skip-install     skip package-manager install after scaffolding
  --list-stacks      print available stack templates
  --show-stack SLUG  print readme + metadata for one stack
  --list-recipes     print registered scratch plan recipes
  --slot KEY=VALUE   fill a required recipe slot; may be repeated
  --json             emit machine-readable JSON output where applicable
