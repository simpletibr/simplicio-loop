usage: simplicio-py proposal [-h] [--root ROOT] [--repo-root REPO_ROOT]
                             [--scope-root SCOPE_ROOT]
                             [--context-snapshot-id CONTEXT_SNAPSHOT_ID]
                             [--context-pack-hash CONTEXT_PACK_HASH]
                             [--stack STACK] [--target TARGET]
                             [--criteria CRITERIA] [--constraints CONSTRAINTS]
                             [--dry-run-task] [--verify-only] [--plan PATH]
                             [--json] [--bound-paths BOUND_PATHS]
                             [--mode {auto,integrated,standalone}]
                             [--task-spec PATH | --task-spec-stdin]
                             [--context-snapshot CONTEXT_SNAPSHOT]
                             [--context-pack CONTEXT_PACK]
                             [--execution-context EXECUTION_CONTEXT]
                             [--effect-authorization EFFECT_AUTHORIZATION]
                             [--attempt-id ATTEMPT_ID] [--lease-id LEASE_ID]
                             [--fencing-token FENCING_TOKEN]
                             [--context-handle CONTEXT_HANDLE]
                             [--coordinator-kind COORDINATOR_KIND]
                             [--coordinator-id COORDINATOR_ID]
                             [goal]

positional arguments:
  goal

options:
  -h, --help            show this help message and exit
  --root ROOT
  --repo-root REPO_ROOT
                        declared repository root for the mutation contract
  --scope-root SCOPE_ROOT
                        declared writable scope root inside repo_root
  --context-snapshot-id CONTEXT_SNAPSHOT_ID
                        Mapper snapshot identity
  --context-pack-hash CONTEXT_PACK_HASH
                        Mapper ContextPack identity
  --stack STACK
  --target TARGET
  --criteria CRITERIA
  --constraints CONSTRAINTS
  --dry-run-task        generate the would-be task output without
                        applying/testing
  --verify-only         run SIMPLICIO_TEST_CMD without model generation or
                        repository mutation
  --plan PATH           deprecated alias: use simplicio-dev-cli edit --plan
                        PATH --apply
  --json                emit stable structured task output
  --bound-paths BOUND_PATHS
                        glob limiting which paths the task may change;
                        repeatable
  --mode {auto,integrated,standalone}
  --task-spec PATH      read one exported simplicio.task-spec/v2 JSON document
                        from PATH
  --task-spec-stdin     read one exported simplicio.task-spec/v2 JSON document
                        from stdin
  --context-snapshot CONTEXT_SNAPSHOT
                        canonical Mapper ContextSnapshot JSON path
  --context-pack CONTEXT_PACK
                        canonical Mapper ContextPack JSON path
  --execution-context EXECUTION_CONTEXT
                        Mapper execution-context/v1 provenance JSON path
  --effect-authorization EFFECT_AUTHORIZATION
                        coordinator-issued EffectAuthorization JSON path
  --attempt-id ATTEMPT_ID
                        coordinator-owned atomic attempt ID
  --lease-id LEASE_ID   coordinator-owned lease ID
  --fencing-token FENCING_TOKEN
                        coordinator-owned fencing token
  --context-handle CONTEXT_HANDLE
                        canonical snapshot ID bound to the attempt
  --coordinator-kind COORDINATOR_KIND
                        coordinator type recorded in the execution profile
  --coordinator-id COORDINATOR_ID
                        coordinator identity recorded in the execution profile
