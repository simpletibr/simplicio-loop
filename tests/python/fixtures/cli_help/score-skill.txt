usage: simplicio-py score-skill [-h] [--scenario SCENARIO_SOURCES]
                                [--extra-scenario EXTRA_SCENARIO] [--json]
                                [--verbose] [--native] [--python]
                                [skill]

positional arguments:
  skill                 skill/law text file path, or - for stdin (default: -)

options:
  -h, --help            show this help message and exit
  --scenario, -s SCENARIO_SOURCES
                        JSON scenario file path (repeatable); falls back to
                        builtin scenarios
  --extra-scenario EXTRA_SCENARIO
                        inline JSON scenario string (repeatable)
  --json
  --verbose, -v         print per-scenario detail even on success
  --native              force the Rust simplicio binary for this command
  --python              force the Python implementation for this command
